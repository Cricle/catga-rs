# catga-rs Simplified Architecture Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Simplify catga-rs from 15 crates to ~6, with unified message types (3 traits) and a Transport trait abstraction for one-line switching between single-machine and distributed modes.

**Architecture:** Define a new `Transport` trait in catga-core that abstracts all messaging operations. Messages are simplified to 3 traits (Request, Command, Event). Third-party implementations simply implement `Transport` without forking.

**Tech Stack:** Rust, async-trait, tokio

## Global Constraints

- All code must pass `cargo clippy --all-targets --all-features`
- No `unsafe_code` in any crate
- Use existing fn-blanket impls from handler.rs (don't break existing Handler impls)
- Keep feature flags for optional dependencies
- Third-party crates can implement Transport trait without forking catga-core

```rust
/// Compile-time type-safe message type identifier.
/// Each message type gets a unique marker type generated by the derive macro.
pub trait MessageTypeId: 'static {
    const NAME: &'static str;
}

/// Base message trait - simplified from 7 to 3 core traits
pub trait Message: Clone + Send + Sync + 'static {
    fn schema_version(&self) -> u32 { 1 }
    fn priority(&self) -> MessagePriority { MessagePriority::Normal }
}

/// A message that produces a typed response.
pub trait Request: Message {
    type Response: Clone + Send + Sync + 'static;
    type TypeId: MessageTypeId;
}

/// A message that has no response value.
pub trait Command: Message {
    type TypeId: MessageTypeId;
}

/// A message delivered to zero or more subscribers.
pub trait Event: Message {
    type TypeId: MessageTypeId;
}

// Backward-compatible aliases for delayed messages
pub trait DelayedMessage: Message {
    fn scheduled_at(&self) -> Option<std::time::SystemTime> { None }
}

pub trait DelayedRequest: Request + DelayedMessage {}
pub trait DelayedEvent: Event + DelayedMessage {}
impl<T> DelayedRequest for T where T: Request + DelayedMessage {}
impl<T> DelayedEvent for T where T: Event + DelayedMessage {}

#[derive(Clone, Copy, Debug, Default, Deserialize, Eq, PartialEq, Serialize)]
pub enum MessagePriority {
    #[default]
    Normal,
    Low,
    High,
    Critical,
}
```

```rust
use crate::{CatgaResult, Command, Event, MessageTypeId, Request};

/// The abstract central source for sending messages.
/// Uses TypeId for compile-time type-safe routing.
#[async_trait]
pub trait Transport: Send + Sync {
    /// Sends a request and waits for a response.
    async fn send<M: Request>(&self, msg: M) -> CatgaResult<M::Response>;

    /// Sends a command (one-way, no response).
    async fn send_command<M: Command>(&self, cmd: M) -> CatgaResult<()>;

    /// Publishes an event (pub-sub, no response).
    async fn publish<M: Event>(&self, event: M) -> CatgaResult<()>;
}
```

```rust
// crates/catga-local/src/local_transport.rs
use std::collections::HashMap;
use std::sync::Arc;

use async_trait::async_trait;
use tokio::sync::RwLock;

use catga_core::{Command, Event, Request, Transport, CatgaResult, Handler, CommandHandler, EventHandler};

/// In-memory transport for single-machine mode.
/// Useful for development and testing.
pub struct LocalTransport {
    request_handlers: Arc<RwLock<HashMap<&'static str, Box<dyn HandlerAny>>>>,
    command_handlers: Arc<RwLock<HashMap<&'static str, Vec<Box<dyn CommandHandlerAny>>>>>,
    event_handlers: Arc<RwLock<HashMap<&'static str, Vec<Box<dyn EventHandlerAny>>>>>,
}

impl Default for LocalTransport {
    fn default() -> Self {
        Self::new()
    }
}

impl LocalTransport {
    pub fn new() -> Self {
        Self {
            request_handlers: Arc::new(RwLock::new(HashMap::new())),
            command_handlers: Arc::new(RwLock::new(HashMap::new())),
            event_handlers: Arc::new(RwLock::new(HashMap::new())),
        }
    }

    pub async fn register_request<M, H>(&self, handler: H)
    where
        M: Request,
        H: Handler<M> + 'static,
    {
        let mut handlers = self.request_handlers.write().await;
        handlers.insert(M::request_type(), Box::new(handler));
    }

    pub async fn register_command<M, H>(&self, handler: H)
    where
        M: Command,
        H: CommandHandler<M> + 'static,
    {
        let mut handlers = self.command_handlers.write().await;
        handlers.entry(M::command_type()).or_default().push(Box::new(handler));
    }

    pub async fn register_event<M, H>(&self, handler: H)
    where
        M: Event,
        H: EventHandler<M> + 'static,
    {
        let mut handlers = self.event_handlers.write().await;
        handlers.entry(M::event_type()).or_default().push(Box::new(handler));
    }
}

#[async_trait]
impl Transport for LocalTransport {
    async fn send<M: Request>(&self, msg: M) -> CatgaResult<M::Response> {
        let handlers = self.request_handlers.read().await;
        let handler = handlers.get(M::request_type())
            .ok_or_else(|| catga_core::CatgaError::new(catga_core::ErrorCode::NotFound, "no handler"))?;
        handler.handle_request(msg).await
    }

    async fn send_command<M: Command>(&self, cmd: M) -> CatgaResult<()> {
        let handlers = self.command_handlers.read().await;
        if let Some(handler_list) = handlers.get(M::command_type()) {
            for handler in handler_list {
                handler.handle_command(cmd.clone()).await?;
            }
        }
        Ok(())
    }

    async fn publish<M: Event>(&self, event: M) -> CatgaResult<()> {
        let handlers = self.event_handlers.read().await;
        if let Some(handler_list) = handlers.get(M::event_type()) {
            for handler in handler_list {
                handler.handle_event(event.clone()).await?;
            }
        }
        Ok(())
    }
}

// Type-erased handler traits for storage
trait HandlerAny: Send + Sync {
    async fn handle_request<M: Request>(&self, msg: M) -> CatgaResult<M::Response>;
}

#[async_trait]
impl<M, H> HandlerAny for H
where
    M: Request,
    H: Handler<M> + Send + Sync,
{
    async fn handle_request(&self, msg: M) -> CatgaResult<M::Response> {
        Handler::<M>::handle(self, msg).await
    }
}

trait CommandHandlerAny: Send + Sync {
    async fn handle_command<M: Command>(&self, cmd: M) -> CatgaResult<()>;
}

#[async_trait]
impl<M, H> CommandHandlerAny for H
where
    M: Command,
    H: CommandHandler<M> + Send + Sync,
{
    async fn handle_command(&self, cmd: M) -> CatgaResult<()> {
        CommandHandler::<M>::handle(self, cmd).await
    }
}

trait EventHandlerAny: Send + Sync {
    async fn handle_event<M: Event>(&self, event: M) -> CatgaResult<()>;
}

#[async_trait]
impl<M, H> EventHandlerAny for H
where
    M: Event,
    H: EventHandler<M> + Send + Sync,
{
    async fn handle_event(&self, event: M) -> CatgaResult<()> {
        EventHandler::<M>::handle(self, event).await
    }
}
```

```rust
use catga_core::{Command, Event, Request, Transport, CatgaResult, Handler, CommandHandler, EventHandler};
use async_trait::async_trait;

pub struct NatsTransport {
    connection: async_nats::Client,
}

#[async_trait]
impl Transport for NatsTransport {
    async fn send<M: Request>(&self, msg: M) -> CatgaResult<M::Response> {
        // Use nats request/reply pattern
        let subject = format!("requests.{}", M::request_type());
        let payload = serde_json::to_vec(&msg).map_err(|e| ...)?;
        let response = self.connection.request(&subject, payload.into()).await?;
        serde_json::from_slice(&response.payload).map_err(|e| ...)
    }

    async fn send_command<M: Command>(&self, cmd: M) -> CatgaResult<()> {
        // Use nats publish pattern
        let subject = format!("commands.{}", M::command_type());
        let payload = serde_json::to_vec(&cmd).map_err(|e| ...)?;
        self.connection.publish(&subject, payload.into()).await?;
        Ok(())
    }

    async fn publish<M: Event>(&self, event: M) -> CatgaResult<()> {
        // Use nats publish pattern
        let subject = format!("events.{}", M::event_type());
        let payload = serde_json::to_vec(&event).map_err(|e| ...)?;
        self.connection.publish(&subject, payload.into()).await?;
        Ok(())
    }
}
```
